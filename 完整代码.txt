 * 环境：Dev-C++ 5.11，C99

#ifndef STUDENT_H
#define STUDENT_H

#include <stdio.h>

/* 最大学生姓名长度 */
#define MAX_NAME     32

/* 课程数量 */
#define MAX_COURSE   5

/* 最多可保存的学生数量 */
#define MAX_STUDENT  100

/* 数据文件保存路径 */
#define DATA_FILE    "data/students.dat"

/* 学生信息结构体 */
typedef struct {
    char id[12];                 /* 学号 */
    char name[MAX_NAME];         /* 姓名 */

    float scores[MAX_COURSE];    /* 各科成绩 */
    float total;                 /* 总分 */
    float average;               /* 平均分 */
} Student;

/* 排序比较函数指针类型 */
typedef int (*CmpFunc)(const Student *, const Student *);

/* 全局学生数组和学生数量 */
extern Student g_stu[MAX_STUDENT];
extern int g_cnt;

/* 课程名称数组 */
extern const char *COURSE_NAME[MAX_COURSE];

/* ui.c：界面与输入相关函数 */

/* main.c - 程序入口与主菜单循环 */
#include "student.h"
#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>

int main() {
    int choice;          /* 保存用户菜单选择 */
    int running = 1;     /* 控制主循环是否继续运行 */

#ifdef _WIN32
    /* Windows 下创建 data 目录，用于保存数据文件 */
    system("mkdir data 2>nul");

    /* 设置控制台编码为 GBK，方便 Dev-C++ 中文显示 */
    system("chcp 936 > nul");
#endif

    /* 程序启动时自动尝试读取历史数据 */
    printf("\n正在加载数据...\n");
    if (load_from_file(g_stu, &g_cnt, DATA_FILE) == 0)
        printf("已从 %s 加载 %d 条记录。\n", DATA_FILE, g_cnt);
    else
        printf("无历史数据，从空开始。\n");

    /* 主菜单循环 */
    while (running) {
        show_menu();
        choice = read_menu_choice();

        /* 输入不合法时重新显示菜单 */
        if (choice < 0) {
            printf("输入无效，请输入 0-9 或 a。\n");
            continue;
        }

        /* 根据用户输入执行对应功能 */
        switch (choice) {
            case 10:
                fuzzy_find();
                pause_enter();
                continue;
            case 1:
                add_student();
                break;
            case 2:
                show_all(g_stu, g_cnt);
                break;
            case 3:
                find_by_id_menu();
                break;
            case 4:
                modify_student();
                break;
            case 5:
                delete_student();
                break;
            case 6:
                sort_menu();
                break;
            case 7:
                show_stats(g_stu, g_cnt);
                break;
            case 8:
                if (save_to_file(g_stu, g_cnt, DATA_FILE) == 0)
                    printf("已保存到 %s（%d 人）。\n", DATA_FILE, g_cnt);
                else
                    printf("保存失败，请检查 data 目录。\n");
                break;
            case 9:
                if (load_from_file(g_stu, &g_cnt, DATA_FILE) == 0)
                    printf("读取成功，共 %d 人。\n", g_cnt);
                else
                    printf("读取失败或文件不存在。\n");
                break;
            case 0:
                /* 退出前询问是否保存 */
                printf("退出前是否保存？[Y/N]: ");
                {
                    char yn;
                    if (scanf(" %c", &yn) == 1) {
                        clear_buf();
                        yn = (char)toupper((unsigned char)yn);
                        if (yn == 'Y') {
                            save_to_file(g_stu, g_cnt, DATA_FILE);
                            printf("已保存。\n");
                        }
                    } else {
                        clear_buf();
                    }
                }
                running = 0;
                break;
            default:
                printf("无此选项。\n");
        }

        /* 每次操作后暂停，方便用户查看结果 */
        if (running && choice != 0) pause_enter();
    }

    printf("\n感谢使用，再见！\n");
    printf("按回车退出...");
    clear_buf();
    getchar();
    return 0;
}

/* ui.c - 菜单与输入校验 */
#include "student.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>
#include <errno.h>

/* 去掉字符串末尾的空白字符，例如换行、空格、制表符等 */
void trim(char *s) {
    int n;
    if (s == NULL) return;

    n = (int)strlen(s);
    while (n > 0 && isspace((unsigned char)s[n - 1]))
        s[--n] = '\0';
}

/* 读取一整行输入，并自动去掉末尾换行 */
void read_line(const char *prompt, char *buf, int size) {
    if (buf == NULL || size <= 0) return;

    if (prompt) printf("%s", prompt);

    if (fgets(buf, size, stdin) != NULL) {
        size_t len = strlen(buf);

        /* 如果输入超过缓冲区长度，清理残留字符 */
        if (len > 0 && buf[len - 1] != '\n') clear_buf();

        trim(buf);
    } else {
        buf[0] = '\0';
    }
}

/* 读取一行输入，如果直接回车，则保留旧值 */
void read_line_keep(const char *prompt, char *buf, int size, const char *old) {
    if (buf == NULL || size <= 0) return;

    printf("%s[%s] ", prompt, old ? old : "");
    read_line(NULL, buf, size);

    /* 用户直接回车时复制旧值 */
    if (buf[0] == '\0' && old != NULL) {
        strncpy(buf, old, (size_t)size - 1);
        buf[size - 1] = '\0';
    }
}

/* 清空输入缓冲区，避免残留字符影响下一次输入 */
void clear_buf(void) {
    int c = getchar();
    while (c != '\n' && c != EOF) c = getchar();
}

/* 读取指定范围内的整数 */
int read_int(const char *prompt, int lo, int hi) {
    int v;

    printf("%s", prompt);
    if (scanf("%d", &v) != 1) {
        clear_buf();
        return -1;
    }
    clear_buf();

    if (v < lo || v > hi) return -1;
    return v;
}

/* 暂停，等待用户按回车返回菜单 */
void pause_enter(void) {
    printf("\n按回车返回菜单...");
    clear_buf();
    getchar();
}

/* 显示主菜单 */
void show_menu(void) {
    printf("\n============================================\n");
    printf("        欢迎使用学生成绩管理系统 v1.0\n");
    printf("============================================\n");
    printf("  1. 录入学生成绩\n");
    printf("  2. 显示全部学生\n");
    printf("  3. 按学号查找学生\n");
    printf("  4. 修改学生成绩\n");
    printf("  5. 删除学生\n");
    printf("  6. 排序（按总分/学号/姓名）\n");
    printf("  7. 统计（最高/最低/平均/不及格）\n");
    printf("  8. 保存到文件\n");
    printf("  9. 从文件读取\n");
    printf("  a. 模糊查询（姓名关键字）\n");
    printf("  0. 退出系统\n");
    printf("============================================\n");
}

/* 读取菜单选项，支持 0-9 和 a/A */
int read_menu_choice(void) {
    char line[32];
    char *endp;
    long v;

    printf("请输入选项[0-9/a]: ");
    if (fgets(line, sizeof(line), stdin) == NULL) return -1;
    trim(line);

    if (line[0] == '\0') return -1;

    /* a 或 A 表示模糊查询 */
    if ((line[0] == 'a' || line[0] == 'A') && line[1] == '\0') return 10;

    /* 使用 strtol 防止 1abc 这类输入被误判为合法 */
    errno = 0;
    v = strtol(line, &endp, 10);
    while (*endp != '\0' && isspace((unsigned char)*endp)) endp++;

    if (errno != 0 || *endp != '\0' || v < 0 || v > 9) return -1;
    return (int)v;
}

/* student.c - 学生增删改查与显示 */
#include "student.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

/* 全局学生数组 */
Student g_stu[MAX_STUDENT];

/* 当前学生数量 */
int g_cnt = 0;

/* 课程名称 */
const char *COURSE_NAME[MAX_COURSE] = {
    "语文", "数学", "英语", "物理", "化学"
};

/* 计算某个学生的总分和平均分 */
void calc_total_avg(Student *s) {
    int i;
    float sum = 0.0f;

    if (s == NULL) return;

    for (i = 0; i < MAX_COURSE; i++)
        sum += s->scores[i];

    s->total = sum;
    s->average = sum / MAX_COURSE;
}

/* 根据学号查找学生，找到返回下标，找不到返回 -1 */
int find_by_id(const Student *arr, int n, const char *id) {
    int i;

    for (i = 0; i < n; i++)
        if (strcmp(arr[i].id, id) == 0) return i;

    return -1;
}

/* 打印学生表头 */
void print_header(void) {
    printf("\n学号        姓名        性别 年龄 ");
    printf("语文   数学   英语   物理   化学   总分     平均\n");
    printf("--------------------------------------------------------------------------\n");
}

/* 打印一个学生的信息 */
void print_one(const Student *s) {
    int i;

    printf("%-12s%-12s%c    %2d  ",
           s->id, s->name, s->gender, s->age);

    for (i = 0; i < MAX_COURSE; i++)
        printf("%5.1f ", s->scores[i]);

    printf("%7.1f %6.2f\n", s->total, s->average);
}

/* 显示全部学生 */
void show_all(const Student *arr, int n) {
    int i;

    if (n <= 0) {
        printf("暂无学生记录。\n");
        return;
    }

    print_header();
    for (i = 0; i < n; i++)
        print_one(&arr[i]);

    printf("共 %d 人。\n", n);
}

/* 输入单科成绩，要求成绩在 0~100 之间 */
static float input_score(const char *sub) {
    float sc;
    int ok = 0;

    while (!ok) {
        printf("  %s成绩[0-100]: ", sub);

        if (scanf("%f", &sc) != 1) {
            clear_buf();
            printf("  输入无效，请重新输入。\n");
            continue;
        }
        clear_buf();

        if (sc < 0.0f || sc > 100.0f)
            printf("  成绩须在 0~100 之间。\n");
        else {
            ok = 1;
            return sc;
        }
    }

    return 0.0f;
}

/* 添加学生 */
void add_student(void) {
    Student s;
    char buf[64];
    char g;
    int i, age;
    float sc;

    if (g_cnt >= MAX_STUDENT) {
        printf("学生已满（最多 %d 人）。\n", MAX_STUDENT);
        return;
    }

    memset(&s, 0, sizeof(s));

    /* 输入学号，并检查是否重复 */
    while (1) {
        read_line("学号: ", buf, sizeof(buf));
        if (buf[0] == '\0') {
            printf("学号不能为空。\n");
            continue;
        }
        if (find_by_id(g_stu, g_cnt, buf) >= 0)
            printf("学号 %s 已存在，请重新输入。\n", buf);
        else
            break;
    }

    strncpy(s.id, buf, sizeof(s.id) - 1);
    s.id[sizeof(s.id) - 1] = '\0';

    /* 输入姓名 */
    read_line("姓名: ", s.name, MAX_NAME);
    if (s.name[0] == '\0') {
        printf("姓名不能为空。\n");
        return;
    }

    /* 输入性别 */
    printf("性别(M/F): ");
    if (scanf(" %c", &g) != 1) {
        clear_buf();
        printf("性别输入无效。\n");
        return;
    }
    clear_buf();

    g = (char)toupper((unsigned char)g);
    if (g != 'M' && g != 'F') {
        printf("性别须为 M 或 F。\n");
        return;
    }
    s.gender = g;

    /* 输入年龄 */
    printf("年龄: ");
    if (scanf("%d", &age) != 1 || age < 1 || age > 150) {
        clear_buf();
        printf("年龄无效。\n");
        return;
    }
    clear_buf();
    s.age = age;

    /* 输入五门课程成绩 */
    for (i = 0; i < MAX_COURSE; i++) {
        sc = input_score(COURSE_NAME[i]);
        s.scores[i] = sc;
    }

    /* 计算总分和平均分 */
    calc_total_avg(&s);

    /* 保存到全局数组 */
    g_stu[g_cnt++] = s;

    printf("录入成功：%s %s，总分 %.1f，平均 %.2f\n",
           s.id, s.name, s.total, s.average);
}

/* 按学号查找学生 */
void find_by_id_menu(void) {
    char id[16];
    int idx;

    read_line("请输入学号: ", id, sizeof(id));
    idx = find_by_id(g_stu, g_cnt, id);

    if (idx < 0) {
        printf("未找到学号 %s 的学生。\n", id);
        return;
    }

    print_header();
    print_one(&g_stu[idx]);
}

/* 按姓名关键字模糊查询 */
void fuzzy_find(void) {
    char key[MAX_NAME];
    int i, found = 0;

    read_line("请输入姓名关键字: ", key, MAX_NAME);
    if (key[0] == '\0') {
        printf("关键字不能为空。\n");
        return;
    }

    print_header();
    for (i = 0; i < g_cnt; i++) {
        if (strstr(g_stu[i].name, key) != NULL) {
            print_one(&g_stu[i]);
            found = 1;
        }
    }

    if (!found)
        printf("未找到包含 \"%s\" 的学生。\n", key);
    else
        printf("以上为满足条件的记录。\n");
}

/* 修改学生信息 */
void modify_student(void) {
    char id[16];
    char buf[64];
    int idx, i, age;
    float sc;
    Student *p;

    read_line("要修改的学号: ", id, sizeof(id));
    idx = find_by_id(g_stu, g_cnt, id);

    if (idx < 0) {
        printf("未找到学号 %s 的学生。\n", id);
        return;
    }

    p = &g_stu[idx];

    printf("--- 当前信息 ---\n");
    print_header();
    print_one(p);
    printf("--- 修改（直接回车保留原值）---\n");

    /* 修改姓名，直接回车则保留 */
    read_line_keep("姓名", buf, MAX_NAME, p->name);
    strncpy(p->name, buf, MAX_NAME - 1);
    p->name[MAX_NAME - 1] = '\0';

    /* 修改性别 */
    printf("性别[%c] (M/F，回车保留): ", p->gender);
    if (fgets(buf, sizeof(buf), stdin) != NULL) {
        trim(buf);
        if (buf[0] != '\0') {
            buf[0] = (char)toupper((unsigned char)buf[0]);
            if (buf[0] == 'M' || buf[0] == 'F')
                p->gender = buf[0];
        }
    }

    /* 修改年龄 */
    printf("年龄[%d] (回车保留): ", p->age);
    if (fgets(buf, sizeof(buf), stdin) != NULL) {
        trim(buf);
        if (buf[0] != '\0') {
            age = atoi(buf);
            if (age >= 1 && age <= 150)
                p->age = age;
        }
    }

    /* 修改各科成绩 */
    for (i = 0; i < MAX_COURSE; i++) {
        printf("%s[%.1f] (回车保留): ", COURSE_NAME[i], p->scores[i]);
        if (fgets(buf, sizeof(buf), stdin) != NULL) {
            trim(buf);
            if (buf[0] != '\0') {
                sc = (float)atof(buf);
                if (sc >= 0.0f && sc <= 100.0f)
                    p->scores[i] = sc;
                else
                    printf("  成绩超出范围，该项未改。\n");
            }
        }
    }

    /* 修改后重新计算总分和平均分 */
    calc_total_avg(p);

    printf("修改完成。\n");
    print_one(p);
}

/* 删除学生 */
void delete_student(void) {
    char id[16];
    char yn;
    int idx, i;

    read_line("要删除的学号: ", id, sizeof(id));
    idx = find_by_id(g_stu, g_cnt, id);

    if (idx < 0) {
        printf("未找到学号 %s 的学生。\n", id);
        return;
    }

    printf("确认删除 %s %s ? [Y/N]: ", g_stu[idx].name, g_stu[idx].id);
    if (scanf(" %c", &yn) != 1) {
        clear_buf();
        return;
    }
    clear_buf();

    yn = (char)toupper((unsigned char)yn);
    if (yn != 'Y') {
        printf("已取消删除。\n");
        return;
    }

    /* 将后面的学生前移一位，覆盖被删除的学生 */
    for (i = idx; i < g_cnt - 1; i++)
        g_stu[i] = g_stu[i + 1];

    g_cnt--;
    printf("删除成功。\n");
}

/* file_io.c - 文件持久化 */
#include "student.h"
#include <stdio.h>

/* 将学生数据保存到二进制文件 */
int save_to_file(const Student *arr, int n, const char *path) {
    FILE *fp;

    /* 参数合法性检查 */
    if (arr == NULL || path == NULL || n < 0 || n > MAX_STUDENT)
        return -1;

    fp = fopen(path, "wb");
    if (!fp) return -1;

    /* 先写入学生数量 */
    if (fwrite(&n, sizeof(int), 1, fp) != 1) {
        fclose(fp);
        return -1;
    }

    /* 再写入学生数组 */
    if (n > 0 && fwrite(arr, sizeof(Student), (size_t)n, fp) != (size_t)n) {
        fclose(fp);
        return -1;
    }

    if (fclose(fp) != 0) return -1;
    return 0;
}

/* 从二进制文件读取学生数据 */
int load_from_file(Student *arr, int *n, const char *path) {
    FILE *fp;
    int cnt;

    /* 参数合法性检查 */
    if (arr == NULL || n == NULL || path == NULL)
        return -1;

    fp = fopen(path, "rb");
    if (!fp) return -1;

    /* 先读取学生数量 */
    if (fread(&cnt, sizeof(int), 1, fp) != 1) {
        fclose(fp);
        return -1;
    }

    /* 防止文件损坏导致数量异常 */
    if (cnt < 0 || cnt > MAX_STUDENT) {
        fclose(fp);
        return -1;
    }

    /* 读取学生数组 */
    if (cnt > 0) {
        if (fread(arr, sizeof(Student), (size_t)cnt, fp) != (size_t)cnt) {
            fclose(fp);
            return -1;
        }
    }

    *n = cnt;
    fclose(fp);
    return 0;
}

/* stats.c - 排序与统计 */
#include "student.h"
#include <stdio.h>
#include <string.h>

/* 按总分降序比较 */
int cmp_by_total(const Student *a, const Student *b) {
    if (a->total < b->total) return 1;
    if (a->total > b->total) return -1;
    return 0;
}

/* 按学号升序比较 */
int cmp_by_id(const Student *a, const Student *b) {
    return strcmp(a->id, b->id);
}

/* 按姓名字典序比较 */
int cmp_by_name(const Student *a, const Student *b) {
    return strcmp(a->name, b->name);
}

/* 冒泡排序，根据传入的比较函数决定排序规则 */
void sort_students(Student *arr, int n, CmpFunc cmp) {
    int i, j;
    Student tmp;

    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - 1 - i; j++) {
            if (cmp(&arr[j], &arr[j + 1]) > 0) {
                tmp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = tmp;
            }
        }
    }
}

/* 排序菜单 */
void sort_menu(void) {
    int key;

    if (g_cnt <= 0) {
        printf("暂无数据可排序。\n");
        return;
    }

    printf("1-按总分降序  2-按学号升序  3-按姓名字典序\n");
    key = read_int("请选择[1-3]: ", 1, 3);

    if (key < 0) {
        printf("选项无效。\n");
        return;
    }

    if (key == 1)
        sort_students(g_stu, g_cnt, cmp_by_total);
    else if (key == 2)
        sort_students(g_stu, g_cnt, cmp_by_id);
    else
        sort_students(g_stu, g_cnt, cmp_by_name);

    printf("排序完成。\n");
    show_all(g_stu, g_cnt);
}

/* 显示成绩统计信息 */
void show_stats(const Student *arr, int n) {
    int i, j, pass, excellent, fail_cnt;
    float minv, maxv, sum, avg;

    if (n <= 0) {
        printf("暂无数据。\n");
        return;
    }

    printf("\n======== 各科统计 ========\n");

    /* 统计每门课程的最高分、最低分、平均分、及格率、优秀率 */
    for (j = 0; j < MAX_COURSE; j++) {
        minv = maxv = arr[0].scores[j];
        sum = 0.0f;
        pass = excellent = 0;

        for (i = 0; i < n; i++) {
            float sc = arr[i].scores[j];

            if (sc < minv) minv = sc;
            if (sc > maxv) maxv = sc;
            sum += sc;

            if (sc >= 60.0f) pass++;
            if (sc >= 85.0f) excellent++;
        }

        avg = sum / n;

        printf("%s: 最高%.1f 最低%.1f 平均%.2f 及格率%.1f%% 优秀率%.1f%%\n",
               COURSE_NAME[j], maxv, minv, avg,
               100.0f * pass / n, 100.0f * excellent / n);
    }

    /* 统计平均分不及格人数 */
    fail_cnt = 0;
    for (i = 0; i < n; i++)
        if (arr[i].average < 60.0f) fail_cnt++;

    printf("\n平均分不及格人数: %d / %d\n", fail_cnt, n);

    printf("\n======== 总分分数段（按平均分） ========\n");

    {
        int seg[5] = {0, 0, 0, 0, 0};

        /* 按平均分统计分数段 */
        for (i = 0; i < n; i++) {
            float a = arr[i].average;

            if (a < 60.0f)
                seg[0]++;
            else if (a < 70.0f)
                seg[1]++;
            else if (a < 80.0f)
                seg[2]++;
            else if (a < 90.0f)
                seg[3]++;
            else
                seg[4]++;
        }

        /* 用 # 简单显示柱状统计结果 */
        printf("[0,60)  : ");
        for (j = 0; j < seg[0]; j++) printf("#");
        printf("  (%d)\n", seg[0]);

        printf("[60,70) : ");
        for (j = 0; j < seg[1]; j++) printf("#");
        printf("  (%d)\n", seg[1]);

        printf("[70,80) : ");
        for (j = 0; j < seg[2]; j++) printf("#");
        printf("  (%d)\n", seg[2]);

        printf("[80,90) : ");
        for (j = 0; j < seg[3]; j++) printf("#");
        printf("  (%d)\n", seg[3]);

        printf("[90,100]: ");
        for (j = 0; j < seg[4]; j++) printf("#");
        printf("  (%d)\n", seg[4]);
    }
}
